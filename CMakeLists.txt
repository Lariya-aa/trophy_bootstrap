cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
 set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(trophy_bootstrap VERSION 0.1.0 LANGUAGES C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_STANDARD 99)
add_executable(trophy_bootstrap src/plugin.c src/bootstrap.c src/sha256.c src/app_list.c src/app_status.c)
target_include_directories(trophy_bootstrap PRIVATE include)
target_compile_definitions(trophy_bootstrap PRIVATE TB_VITA=1)
target_compile_options(trophy_bootstrap PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-misleading-indentation -O2 -fno-lto -nostdlib)
target_link_options(trophy_bootstrap PRIVATE -nostdlib -Wl,-q)
target_link_libraries(trophy_bootstrap k gcc SceLibKernel_stub SceAppMgr_stub SceIofilemgr_stub SceRtc_stub SceKernelModulemgr_stub SceKernelThreadMgr_stub SceProcessmgr_stub)
vita_create_self(trophy_bootstrap.suprx trophy_bootstrap CONFIG exports.yml UNSAFE)
